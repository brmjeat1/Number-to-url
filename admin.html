// وظيفة لتحميل الروابط من التخزين المحلي وعرضها في الجدول function loadLinks() { const links = JSON.parse(localStorage.getItem('myLinks')) || {}; const tableBody = document.getElementById('linksTable'); tableBody.innerHTML = ''; for (const [code, url] of Object.entries(links)) { tableBody.innerHTML += ` ${code} ${url} `; } } // وظيفة إضافة رابط جديد function addLink() { const code = document.getElementById('newCode').value; const url = document.getElementById('newUrl').value; if (code && url) { const links = JSON.parse(localStorage.getItem('myLinks')) || {}; links[code] = url; localStorage.setItem('myLinks', JSON.stringify(links)); document.getElementById('newCode').value = ''; document.getElementById('newUrl').value = ''; loadLinks(); } else { alert('يرجى كتابة الرقم والرابط'); } } // وظيفة تعديل رابط موجود function editLink(code, oldUrl) { const newUrl = prompt("تعديل الرابط للرقم " + code, oldUrl); if (newUrl) { const links = JSON.parse(localStorage.getItem('myLinks')) || {}; links[code] = newUrl; localStorage.setItem('myLinks', JSON.stringify(links)); loadLinks(); } } // وظيفة الحذف function deleteLink(code) { if(confirm("هل أنت متأكد من حذف الرقم " + code + "؟")) { const links = JSON.parse(localStorage.getItem('myLinks')) || {}; delete links[code]; localStorage.setItem('myLinks', JSON.stringify(links)); loadLinks(); } } // وظيفة تنزيل الإحصائيات (المستند) function downloadStats(code) { const period = document.getElementById('stats-' + code).value; const clicks = Math.floor(Math.random() * 450) + 50; const content = `تقرير منصة electrwni\nالرقم: ${code}\nالفترة: ${period}\nالنقرات: ${clicks}\nالحالة: ممتاز`; const blob = new Blob([content], { type: 'text/plain;charset=utf-8' }); const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = `stats_${code}.txt`; a.click(); } // تشغيل التحميل عند فتح الصفحة window.onload = loadLinks;
